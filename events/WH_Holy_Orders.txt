namespace = WH_holy_orders

#Knights of Sigmar's Blood

WH_holy_orders.01 = {
	type = character_event	
	
	hidden = yes

	
	immediate = {
		every_living_character = {
			limit = {
				any_held_title = {
					this = title:d_sigmar_blood_knights
				}
			}
			save_scope_as = leader
		}
		hidden_effect = {
			if = {
				limit = { NOT = { title:b_gablitz.holder = scope:leader } }
				create_title_and_vassal_change = {
					type = leased_out
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:b_gablitz = {
					change_title_holder_include_vassals = {
						holder = scope:leader
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		create_holy_order = {
			leader = scope:leader
			capital = title:b_gablitz
			save_scope_as = sigmar_blood_knights
		}

		set_global_variable = {
			name = sigmar_blood_knights_title
			value = scope:sigmar_blood_knights.title
		}
	
		hidden_effect = {
			scope:leader = {
				add_gold = 100 #So that they have some money to lend out
				add_piety_level = 2
				add_gold = holy_order_starting_gold
				every_courtier = {
					add_trait = order_member
				}
			}
		}
		global_var:sigmar_blood_knights_title = {
			copy_title_history = title:d_sigmar_blood_knights
		}
		
		title:b_gablitz = {
			lease_out_to = scope:sigmar_blood_knights
		}		

		create_holy_order_effect = yes

	}
}
