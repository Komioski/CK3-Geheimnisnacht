nubile_age = {
	if = {
		limit = { 
			is_human = yes
		}
		value = 16
	}
	else_if = {
		limit = { 
			OR = {
				is_orc = yes 
				cannot_be_killed = yes
			}
		}
		value = 16
	}
	else_if = { # 70
		limit = { is_dwarf = yes }
		value = 70
	}
	# else_if = { # 33
	# 	limit = { is_hobbit = yes } #TODO WH? Halfling?

years_to_nubility = {
	value = nubile_age
	subtract = age
	min = 0
}

additional_years_of_fertility = {
	value = 0
	if = {
		limit = { is_human = yes }
	}
	else_if = {
		limit = {
			OR = {
				is_orc = yes
				cannot_be_killed = yes
			}
		}
		add = 80
	}
	else_if = {
		limit = { is_dwarf = yes }
		add = 130
	}
	# else_if = {
	# 	limit = { is_hobbit = yes } #TODO WH? Halfling?
	# 	add = 20
	# }
	else_if = {
		limit = { is_elf = yes }
		add = 13000
	}
	
	if = {
		limit = { has_trait = fecund }
		add = 5
	}
	# if = {
	# 	limit = { has_character_modifier = reached_meneltarma }
	# 	add = 20
	# }
	if = {
		limit = {
			exists = dynasty
			dynasty = {
				has_dynasty_perk = blood_legacy_5
			}
		}
		add = 5
	}
}

additional_life_expectancy = {
	value = 0
	if = {
		limit = { is_human = yes }
		# if = { # Can only be gained by humans
		# 	limit = { has_trait = skinchanger }
		# 	add = 30
		# }

	}
	else_if = {
		limit = {
			OR = {
				is_orc = yes
				cannot_be_killed = yes
			}
		}
		add = 120
	}
	else_if = {
		limit = { is_dwarf = yes }
		add = 170
	}
	# else_if = {
	# 	limit = { is_hobbit = yes } #TODO WH? Halfling?
	# 	add = 30
	# }
	else_if = {
		limit = {
			OR = {
				cannot_be_killed = yes
				is_elf = yes
			}
		}
		add = 13000
	}
	
	if = {
		limit = { has_trait = fecund }
		add = 5
	}
	if = {
		limit = { has_character_modifier = dark_vitality_modifier }
		add = 20
	}
	if = {
		limit = { has_character_modifier = radagast_advice_4_modifier }
		add = 3
	}
}

young_age = { # Equivalent to human 25 years old
	if = {
		limit = { is_elf = yes }
		value = 300
	}
	else_if = {
		limit = { cannot_be_killed = yes }
		value = 25
	}
	else = {
		value = total_life_expectancy
		divide = 3.2
	}
}

middle_age = { # Equivalent to human 40 years old
	if = {
		limit = { is_elf = yes }
		value = 3000
	}
	else_if = {
		limit = { cannot_be_killed = yes }
		value = 40
	}
	else = {
		value = total_life_expectancy
		divide = 2
	}
}	

old_age = { # Equivalent to human 60 years old
	if = {
		limit = { is_elf = yes }
		value = 7000
	}
	else_if = {
		limit = { cannot_be_killed = yes }
		value = 60
	}
	else = {
		value = 60
		add = additional_life_expectancy
	}
}
