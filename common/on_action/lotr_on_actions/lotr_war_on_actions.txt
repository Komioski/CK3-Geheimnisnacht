lotr_on_join_war_as_secondary = {
	effect = {
		if = { # Haven’t we done this before? | Achievement
			limit = {
				NOT = { exists = global_var:lotr_achievement_12 }
				realms_achievements_enabled = yes
				OR = {
					is_ai = no
					scope:war = { primary_attacker = { is_ai = no } }
				}
				trigger_if = {
					limit = { is_human = yes }
					scope:war = {
						any_war_attacker = {
							is_elf = yes
						}
						any_war_attacker = {
							is_dwarf = yes
						}
					}
				}
				trigger_else_if = {
					limit = { is_elf = yes }
					scope:war = {
						any_war_attacker = {
							is_human = yes
						}
						any_war_attacker = {
							is_dwarf = yes
						}
					}
				}
				trigger_else_if = {
					limit = { is_dwarf = yes }
					scope:war = {
						any_war_attacker = {
							is_human = yes
						}
						any_war_attacker = {
							is_elf = yes
						}
					}
				}
				trigger_else = { 
					scope:war = {
						any_war_attacker = {
							is_human = yes
						}
						any_war_attacker = {
							is_elf = yes
						}
						any_war_attacker = {
							is_dwarf = yes
						}
					}
				}
			}
			set_global_variable = lotr_achievement_12
		}
	}
}
lotr_on_war_started = {
	events = {
		wastelands.0002 # Immediate truce if declaring war agains Wastelands
	}
	effect = {
		# Struggle catalyst #LotR
		scope:attacker = {
			if = {
				limit = {
					any_character_struggle = {
						involvement = involved
						phase_has_catalyst = catalyst_declared_a_war
					}
				}
				every_character_struggle = {
					involvement = involved
					activate_struggle_catalyst = {
						catalyst = catalyst_declared_a_war
						character = root
					}
				}
			}
		}
		scope:defender = {
			every_tributary = {
				limit = { subject_contract_has_flag = tributary_contract_tributary_forced_war_override }
				trigger_event = {
					id = tributary.0008
					days = 1
				}
			}
			#trigger_event = { #Juke note: Unused
			#	id = tributary.0001
			#	days = 1
			#}
		}
		
		#Defensive Alliance variable tracker. Confederation members get automatically added, so might as well add this
		scope:defender = {
			if = {
				limit = {
					is_confederation_member = yes
					confederation ?= { has_variable = defensive_alliance }
				}
				confederation ?= { increase_variable = { NAME = lotr_defensive_wars_called_into AMOUNT = 1 } }
			}
		}
	}
}

lotr_on_war_won_attacker = {
	effect = {
		if = {
			limit = {
				scope:attacker = { is_ai = no } 
				realms_achievements_enabled = yes
			}
			if = { # The Old, the True | Achievement
				limit = { 
					NOT = { exists = global_var:lotr_achievement_71 }
					scope:attacker.dynasty = dynasty:dynasty_khadnom
					scope:attacker = { is_tributary = yes }
					scope:defender = scope:attacker.suzerain
					has_title = title:h_harad
				}
				set_global_variable = lotr_achievement_71
			}
			if = { # Fatherly Ire achievement
				limit = {
					NOT = { exists = global_var:lotr_achievement_31 }
					scope:attacker = character:lineofelrond1
					scope:defender = character:lineisildur42
				}
				set_global_variable = lotr_achievement_31
			}
		}
		if = {
			limit = {
				scope:attacker = {
					has_title = title:e_yellow_mountains
					is_orc = yes
					is_independent_ruler = yes 
				}
			}
			if = { # Yellow Mountain Goblin Rebellion: Bruka Victory
				limit = {
					NOR = { 
						exists = global_var:lotr_YM_GoblinRebellion_variable1
						exists = global_var:lotr_YM_GoblinRebellion_variable2
					}
					scope:attacker = character:lineofgrishok79
					scope:defender = character:lineoflughasya23
				}
				set_global_variable = lotr_YM_GoblinRebellion_variable1
				set_global_variable = lotr_YM_GoblinRebellion_variable2
				title:e_yellow_mountains.holder = {
					trigger_event = {
						id = yellow_mountain_orc.006
						days = 1 
					}
				}				
			}		
			if = { # Yellow Mountain Goblin Rebellion: Ozag Victory
				limit = {
					NOR = { 
						exists = global_var:lotr_YM_GoblinRebellion_variable1
						exists = global_var:lotr_YM_GoblinRebellion_variable2
					}
					scope:attacker = character:lineofzukarok10
					scope:defender = character:lineoflughasya23
				}
				set_global_variable = lotr_YM_GoblinRebellion_variable1
				set_global_variable = lotr_YM_GoblinRebellion_variable2
				title:e_yellow_mountains.holder = {
					trigger_event = {
						id = yellow_mountain_orc.007
						days = 1 
					}
				}				
			}		
		}
		if = { # Seregul loses Pél explosion
			limit = {
				scope:loser = character:lineofagathohin173
				global_var:pel_destabilized = yes
				scope:war = { using_cb = nation_fracturing_faction_war }
			}
			character:lineofagathohin173 = { 
				trigger_event = {
					id = seregul.0254 
					days = 1
				}
			}
		}
		if = { # Seregul exposed and defeated
			limit = {
				scope:loser = character:lineofagathohin173
				scope:winner = title:d_balan_lai_leader.holder
				global_var:seregul_exposed = yes
			}
			scope:winner = {
				trigger_event = seregul.0415
				imprison = {
					target = scope:loser
					type = dungeon
				}
				trigger_event = {
					id = seregul.0417
					days = 1
				}
			}
		}
	}
}
